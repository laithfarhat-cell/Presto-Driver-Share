// ══════ منظومة عروض وساعات المدن — كود الربط v3 (الصقه بالكامل مكان القديم) ══════
// الجديد: تبويب مستقل لكل مدينة + تحديث العرض في كل التبويبات التي يمتد عليها (مثل عروض كل طرابلس)
var HEADERS = ['التاريخ','وقت التفعيل','المشرف','المدينة','المنطقة','رقم المنطقة','الخدمات','قيمة العرض الحالية','القيمة عند التفعيل','عدد السائقين','السبب','عدد التخفيضات','سجل التخفيضات','وقت الإيقاف','إجمالي الساعات','الحالة','صورة مرفقة','معرّف السجل'];
var COLORS = { active:'#e6f4ea', reduced:'#fef7e0', stopped:'#fce8e6' };
var DEFAULT_SHEET = 'Presto Driver Share';

function doGet(e){ return handle(e && e.parameter ? e.parameter : {}); }

function doPost(e){
  var data = {};
  try{ data = JSON.parse(e.postData.contents); }catch(err){ data = {}; }
  return handle(data);
}

function handle(d){
  var lock = LockService.getScriptLock();
  try{ lock.waitLock(25000); }catch(e){}
  try{
    if(!d.sheetId){ return out({ok:false, error:'لا يوجد معرّف ملف'}); }
    var ss = SpreadsheetApp.openById(d.sheetId);
    var action = d.action || 'test';
    if(action === 'test' || action === 'init'){
      var sheet = getOrCreateSheet(ss, cleanName(d.sheetName) || DEFAULT_SHEET);
      return out({ok:true, msg:'تم الاتصال', sheet: sheet.getName()});
    }
    if(action === 'create'){ return out(createByCity(ss, d)); }
    if(action === 'update'){ return out(updateEverywhere(ss, d)); }
    return out({ok:false, error:'إجراء غير معروف'});
  }catch(err){
    return out({ok:false, error:String(err)});
  }finally{
    try{ lock.releaseLock(); }catch(e){}
  }
}

function cleanName(s){
  return String(s||'').replace(/[\[\]\*\?\:\/\\]/g,' ').replace(/\s+/g,' ').trim().slice(0,80);
}

function getOrCreateSheet(ss, name){
  var sheet = ss.getSheetByName(name);
  if(!sheet){ sheet = ss.insertSheet(name); }
  ensureHeaders(sheet);
  return sheet;
}

function out(obj){
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function ensureHeaders(sheet){
  if(sheet.getLastRow() === 0){
    sheet.appendRow(HEADERS);
    sheet.getRange(1,1,1,HEADERS.length).setFontWeight('bold').setBackground('#111111').setFontColor('#ff8a00').setHorizontalAlignment('center');
    sheet.setFrozenRows(1);
    sheet.setRightToLeft(true);
    sheet.setColumnWidth(13, 240);
    sheet.setColumnWidth(17, 130);
  }
}

// إنشاء: توزيع صفوف العرض على تبويبات المدن (كل مدينة ← تبويبها)
function createByCity(ss, d){
  var groups = {};
  var order = [];
  for(var i=0; d.zones.length > i; i++){
    var z = d.zones[i];
    var city = cleanName(z.city) || DEFAULT_SHEET;
    if(!groups[city]){ groups[city] = []; order.push(city); }
    groups[city].push(z);
  }
  var made = 0;
  var tabs = [];
  for(var c=0; order.length > c; c++){
    var cityName = order[c];
    var sheet = getOrCreateSheet(ss, cityName);
    var existing = existingKeys(sheet);
    var list = groups[cityName];
    for(var j=0; list.length > j; j++){
      var z2 = list[j];
      var key = d.recordId + '|' + z2.name;
      if(existing[key]){ continue; }
      sheet.appendRow([ d.date, d.startTime, d.supervisor, z2.city, z2.name, z2.id || '—',
        d.services.join(' + '), d.value, d.originalValue, d.drivers || '', d.reason || '',
        0, '', '', '', 'نشط', d.image ? '📎' : '', d.recordId ]);
      var row = sheet.getLastRow();
      sheet.getRange(row,1,1,HEADERS.length).setBackground(COLORS.active);
      if(d.image){
        try{
          var blob = Utilities.newBlob(Utilities.base64Decode(d.image), 'image/jpeg', 'offer.jpg');
          sheet.setRowHeight(row, 95);
          sheet.insertImage(blob, 17, row);
        }catch(imgErr){}
      }
      made++;
    }
    tabs.push(cityName);
  }
  return {ok:true, created:made, tabs:tabs};
}

// تحديث شامل: يمر على كل التبويبات ويحدّث كل صف يحمل معرّف العرض (لضمان تحديث عروض كل طرابلس والعروض الممتدة لأكثر من مدينة)
function updateEverywhere(ss, d){
  var sheets = ss.getSheets();
  var updated = 0;
  for(var i=0; sheets.length > i; i++){
    var res = updateRows(sheets[i], d);
    updated += res.updated;
  }
  return {ok:true, updated:updated};
}

function existingKeys(sheet){
  var map = {};
  var last = sheet.getLastRow();
  if(2 > last){ return map; }
  var ids = sheet.getRange(2,18,last-1,1).getValues();
  var zones = sheet.getRange(2,5,last-1,1).getValues();
  for(var i=0; ids.length > i; i++){ map[ids[i][0] + '|' + zones[i][0]] = true; }
  return map;
}

function updateRows(sheet, d){
  var last = sheet.getLastRow();
  var updated = 0;
  if(2 > last){ return {ok:true, updated:0}; }
  var ids = sheet.getRange(2,18,last-1,1).getValues();
  for(var i=0; ids.length > i; i++){
    if(String(ids[i][0]) === String(d.recordId)){
      var row = i + 2;
      sheet.getRange(row, 8).setValue(d.value);
      sheet.getRange(row, 12).setValue(d.reductionsCount);
      sheet.getRange(row, 13).setValue(d.reductionsLog || '');
      sheet.getRange(row, 14).setValue(d.stopTime || '');
      sheet.getRange(row, 15).setValue(d.totalHours || '');
      sheet.getRange(row, 16).setValue(d.statusText);
      sheet.getRange(row,1,1,HEADERS.length).setBackground(COLORS[d.statusKey] || COLORS.active);
      updated++;
    }
  }
  return {ok:true, updated:updated};
}
